async function publicFont(route) {
  const request = route.request();
  const url = request.url();
  const css = url === cssUrl;
  const entry = { url, method: request.method(), css, status: null, startedAt: new Date().toISOString() };
  fontResponses.push(entry);
  try {
    if (request.method() !== 'GET') throw Error('Font request must be GET');
    const userAgent = await request.headerValue('user-agent');
    const response = await fetch(url, { redirect: 'manual', credentials: 'omit',
      headers: { 'user-agent': userAgent, accept: css ? 'text/css,*/*;q=0.1' : '*/*' }, signal: AbortSignal.timeout(15000) });
    entry.status = response.status;
    entry.headers = Object.fromEntries(['content-type', 'cache-control', 'etag', 'access-control-allow-origin', 'vary', 'cross-origin-resource-policy'].map(k => [k, response.headers.get(k)]).filter(([,v]) => v !== null));
    if (response.status !== 200) throw Error(`Public font HTTP${response.status}; redirects not followed`);
    const limit = css ? 256 * 1024 : 2 * 1024 * 1024;
    const chunks = []; let bytes = 0;
    for await (const chunk of response.body) { bytes += chunk.byteLength; if (bytes > limit) throw Error('Public font body cap exceeded'); chunks.push(chunk); }
    const body = Buffer.concat(chunks);
    entry.bytes = body.byteLength; entry.sha256 = digest(body);
    const contentType = response.headers.get('content-type') ?? '';
    if (css && !contentType.startsWith('text/css')) throw Error('Exact CSS URL did not serve CSS');
    if (!css && !/(?:font|woff|octet-stream)/i.test(contentType)) throw Error('Referenced font URL did not serve font bytes');
    entry.file = `fonts/${entry.sha256}${css ? '.css.txt' : '.woff2'}`;
    const destination = path.join(out, entry.file);
    if (fs.existsSync(destination) && digest(fs.readFileSync(destination)) !== entry.sha256) throw Error('Font payload collision');
    if (!fs.existsSync(destination)) fs.writeFileSync(destination, body);
    if (css) {
      const references = [...body.toString('utf8').matchAll(/url\(\s*['"]?(https:\/\/[^)'"\s]+)['"]?\s*\)/g)].map(m => m[1]);
      if (!references.length) throw Error('Public CSS has no font references');
      for (const href of references) { const u = new URL(href); if (u.origin !== 'https://fonts.gstatic.com' || u.username || u.password || !u.pathname.endsWith('.woff2')) throw Error('Unapproved font reference in public CSS'); allowedFonts.add(href); }
      entry.references = references;
    }
    await route.fulfill({ status: response.status, headers: entry.headers, body });
    entry.fulfilledExactBytes = true;
  } catch (error) {
    entry.error = String(error);
    report.events.push({ page: 'font-recorder', phase, type: 'font-network-error', url, text: String(error) });
    await route.abort('failed');
  } finally { entry.finishedAt = new Date().toISOString(); put('font-network.json', { exactCssUrl: cssUrl, allowedReferencedFonts: [...allowedFonts].sort(), responses: fontResponses }); }
}
async function usedFonts(page, name) {
  const fontFaceSet = await page.evaluate(() => ({ status: document.fonts.status,
    declared: [...document.fonts].map(f => ({ family: f.family, weight: f.weight, style: f.style, status: f.status })),
    bodyFamily: getComputedStyle(document.body).fontFamily }));
  const client = await page.context().newCDPSession(page);
  const actual = [];
  try {
    await client.send('DOM.enable'); await client.send('CSS.enable');
    const { root } = await client.send('DOM.getDocument');
    for (const selector of ['h1,h2', 'p', '.font-mono', 'pre']) {
      const { nodeId } = await client.send('DOM.querySelector', { nodeId: root.nodeId, selector });
      if (nodeId) actual.push({ selector, ...(await client.send('CSS.getPlatformFontsForNode', { nodeId })) });
    }
  } finally { await client.detach(); }
  put(name + '.fonts.json', { fontFaceSet, actual });
  if (fontFaceSet.status !== 'loaded' || !fontFaceSet.declared.some(f => f.family.replaceAll('"','') === 'Inter' && f.status === 'loaded') || !fontFaceSet.declared.some(f => f.family.replaceAll('"','') === 'JetBrains Mono' && f.status === 'loaded')) throw Error('Both declared public font families must actually be loaded before capture');
}

async function watch(page, name) {
  page.on('console', m => { if (['error', 'warning'].includes(m.type())) report.events.push({ page: name, phase, type: m.type(), text: m.text() }); });
  page.on('pageerror', e => report.events.push({ page: name, phase, type: 'pageerror', text: String(e) }));
  page.on('request', r => report.requests.push({ page: name, phase, method: r.method(), url: r.url() }));
  page.on('requestfailed', r => report.events.push({ page: name, phase, type: 'requestfailed', url: r.url(), error: r.failure(),
    intentionalTransportClose: (closing || navigating.has(name)) && r.resourceType() === 'eventsource' }));
  page.on('response', r => {
    const event = { page: name, phase, status: r.status(), url: r.url(), method: r.request().method() };
    report.responses.push(event);
    if (r.status() >= 400) report.events.push({ ...event, type: 'http-error' });
  });
  return page;
}

}

}

  report.explicitStartNewErrors = failed.utterances.filter(u => u.speechAct === 'error' && !priorErrors.has(u.id));
  check('one explicit Start adds one new missing-key error', report.explicitStartNewErrors.length === 1);
  await expect(joiner.getByText(/turn failed:.*OPENAI_API_KEY is not configured/).first()).toBeVisible();
  await capture(joiner, '1440-no-key-halted'); await capture(lobby, '390-no-key-halted');
  await joiner.getByRole('button', { name: 'State', exact: true }).click();
  await capture(joiner, '1440-no-key-state');
  navigating.add('joiner');
  await joiner.goto(base + '/', { waitUntil: 'networkidle' });
  report.actions.push({ phase, action: 'Explicit same-context route navigation to root after failure; not a discovered in-app Back control' });
  navigating.delete('joiner');
  await expect(joiner.getByRole('button', { name: 'Create room', exact: true })).toBeVisible();
  await capture(joiner, '1440-lobby-recovery');
  await apiRoom(id, 'retained-room-after-navigation');
  await demo.setViewportSize({ width: 1440, height: 960 });
  await demo.bringToFront();
  phase = 'count100';
  const responsePromise = demo.waitForResponse(r => r.url() === base + '/compare/demo' && r.request().method() === 'POST', { timeout: 300000 });
  await demo.getByRole('button', { name: /Run the comparison/i }).click();
  report.actions.push({ phase, action: 'One native Run the comparison click at shipped100/100 deterministic defaults' });
  const started = Date.now();
  const apiResponse = await responsePromise;
  const api = await apiResponse.json();
  put('same-run-api.json', { status: apiResponse.status(), request: apiResponse.request().postDataJSON(), body: api });

  check('three bad private states remain1', api.bad?.at(-1)?.agentStates?.length === 3 && api.bad.at(-1).agentStates.every(s => s.believesCurrent === 1));
  const progressRe = /\b(\d{1,3})\s*\/\s*100\b/;
  let progress = null;
  while (Date.now() - started < 300000) {
    const body = await demo.evaluate(() => document.body.innerText);
    const match = body.match(progressRe);
    const current = match ? Number(match[1]) : null;
    if (current !== progress) { progress = current; report.samples.push({ current, atMs: Date.now() - started }); persist(); }
    if (current === 100) { report.firstHundredMs = Date.now() - started; break; }
    await demo.waitForTimeout(1000);
  }
  check('native rendered100 within original300s deadline', progress === 100);

  phase = 'current-completed-matrix';
  for (const [width, height] of pairs) {

  check('only exact public CSS/referenced fonts permitted externally', report.blockedEgress.length === 0 && fontResponses.length > 0 && fontResponses.every(r => r.status === 200 && r.fulfilledExactBytes));
  check('no unexpected browser console/request/HTTP errors', report.events.filter(e => !e.intentionalTransportClose).length === 0);
  report.status = 'PASS';
} catch (error) {
  report.status = 'FAIL'; report.failures.push({ phase, error: String(error), stack: error.stack });
  for (const c of contexts) for (const page of c.pages()) {
    try { await capture(page, `failure-${contexts.indexOf(c)}-${c.pages().indexOf(page)}`); } catch (captureError) { report.failures.push({ phase: 'failure-capture', error: String(captureError) }); }
  }
} finally {
  closing = true;
  for (const c of contexts) { try { await c.close(); } catch (e) { report.failures.push({ phase: 'context-close', error: String(e) }); } }
  if (browser) { try { await browser.close(); } catch (e) { report.failures.push({ phase: 'browser-close', error: String(e) }); } }
  const unexpected = report.events.filter(e => !e.intentionalTransportClose);
  if (unexpected.length || report.blockedEgress.length) report.status = 'FAIL';
  report.finishedAt = new Date().toISOString(); report.closedOwnedBrowser = true; persist();
  console.log(JSON.stringify({ status: report.status, checks: report.checks.length, captures: report.captures.length, firstHundredMs: report.firstHundredMs, failures: report.failures, browserEvents: report.events.length }));
  if (report.status !== 'PASS' || report.failures.length) process.exitCode = 1;
}
