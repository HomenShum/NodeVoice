from pathlib import Path
import json,hashlib,subprocess,os,time
S=Path(__file__).resolve().parent;C=S/'consumer';E=S/'evidence'

sha=lambda b:hashlib.sha256(b).hexdigest()
def write(p,j):p.write_text(json.dumps(j,indent=2,ensure_ascii=False)+'\n',encoding='utf-8',newline='\n')
baseline=json.loads(originalLock);allowed={r['path'] for r in plan['rows']}
installed=[]
for row in plan['rows']:
    for f in sorted((C/row['path']).rglob('*')):
        if f.is_file():
            assert not f.is_symlink() and not f.lstat().st_file_attributes&1024
            b=f.read_bytes();installed.append({'path':f.relative_to(C).as_posix(),'bytes':len(b),'sha256':sha(b)})
write(E/'installed-baseline-seven-packages.json',installed)
for name in ['browserslist/index.js','browserslist/node.js','browserslist/package.json','nanoid/index.cjs','nanoid/non-secure/index.cjs','nanoid/package.json','postcss/lib/previous-map.js','postcss/lib/input.js','postcss/package.json']:
    f=C/'node_modules'/name;(E/('baseline-'+name.replace('/','__')+'.txt')).write_bytes(f.read_bytes())
env=os.environ.copy()
for key in list(env):
    if key.upper().startswith(('NODE_','OPENAI_','OPENROUTER_','ANTHROPIC_','GEMINI_','GOOGLE_','ELEVENLABS_','OLLAMA_','CONVEX_','VITE_','BROWSERSLIST_')):del env[key]
env.update(SOURCE='deterministic',USE_OLLAMA='0')
commands=[]
def npm(label):
    argv=['C:/nvm4w/nodejs/node.exe','C:/nvm4w/nodejs/node_modules/npm/bin/npm-cli.js','install','--package-lock-only','--ignore-scripts','--no-audit','--no-fund']
    start=time.monotonic()
    with (E/(label+'.log')).open('wb') as out:r=subprocess.run(argv,cwd=C,env=env,stdout=out,stderr=subprocess.STDOUT,timeout=120)
    commands.append({'label':label,'argv':argv,'exitCode':r.returncode,'seconds':round(time.monotonic()-start,3),'logSha256':sha((E/(label+'.log')).read_bytes())})
    write(E/'resolution-commands.json',commands);assert r.returncode==0,label
temporary=json.loads(originalPackage)
for row in plan['rows']:temporary['devDependencies'][row['path'].removeprefix('node_modules/')]=row['targetVersion']
(C/'package.json').write_text(json.dumps(temporary,indent=2,ensure_ascii=False)+'\n',encoding='utf-8',newline='\n')
(E/'temporary-exact-package.json.txt').write_bytes((C/'package.json').read_bytes())

    npm('resolve-temporary-exact')

    assert set(delta)<=allowed|{''},'Unrelated native lock-row churn; preserve and stop'

    (C/'package.json').write_bytes(originalPackage)

final=json.loads((C/'package-lock.json').read_text(encoding='utf-8'))
